CREATE DATABASE IF NOT EXISTS ipl_db;

USE ipl_db;

-- 1. Teams
CREATE TABLE Teams (
    team_id INT AUTO_INCREMENT PRIMARY KEY,
    team_name VARCHAR(100) NOT NULL,
    home_city VARCHAR(50),
    coach_name VARCHAR(100)
);

-- 2. Players
CREATE TABLE Players (
    player_id INT AUTO_INCREMENT PRIMARY KEY,
    player_name VARCHAR(100) NOT NULL,
    role VARCHAR(50),
    nationality VARCHAR(50),
    team_id INT,

    FOREIGN KEY (team_id) REFERENCES Teams(team_id)
);

-- 3. Venues
CREATE TABLE Venues (
    venue_id INT AUTO_INCREMENT PRIMARY KEY,
    venue_name VARCHAR(100) NOT NULL,
    city VARCHAR(50)
);

-- 4. Matches
CREATE TABLE Matches (
    match_id INT AUTO_INCREMENT PRIMARY KEY,
    match_date DATE,
    season YEAR,
    team1_id INT,
    team2_id INT,
    venue_id INT,
    winner_team_id INT,

    FOREIGN KEY (team1_id) REFERENCES Teams(team_id),
    FOREIGN KEY (team2_id) REFERENCES Teams(team_id),
    FOREIGN KEY (venue_id) REFERENCES Venues(venue_id),
    FOREIGN KEY (winner_team_id) REFERENCES Teams(team_id)
);

-- 5. Umpires
CREATE TABLE Umpires (
    umpire_id INT AUTO_INCREMENT PRIMARY KEY,
    umpire_name VARCHAR(100),
    country VARCHAR(50)
);

-- 6. Match_Umpires
CREATE TABLE Match_Umpires (
    assignment_id INT AUTO_INCREMENT PRIMARY KEY,
    match_id INT,
    umpire_id INT,
    umpire_type VARCHAR(30),

    FOREIGN KEY (match_id) REFERENCES Matches(match_id),
    FOREIGN KEY (umpire_id) REFERENCES Umpires(umpire_id)
);

1. Display all players from Mumbai Indians

query
-----
SELECT *
FROM Players
WHERE team_id = 1;

Explanation
------------
SELECT * : SELECT is used to choose data from a table. * means show all columns, so Show every column from the table.
FROM Players : FROM tells SQL from which table the data should come. Players is the table name, so Take the data from the Players table.
WHERE team_id = 1 : WHERE is used to filter rows. team_id = 1 means only choose players whose team id is 1, so Show only players from Mumbai Indians.

query
------
SELECT team_id, COUNT(player_id) AS total_players
FROM Players

GROUP BY team_id : Group rows based on team_id. SQL calculates player count separately for each team.

3. Find the team with the most wins

query
------
SELECT Teams.team_name, COUNT(Matches.match_id) AS total_wins
FROM Teams
JOIN Matches
ON Teams.team_id = Matches.winner_team_id
GROUP BY Teams.team_name
ORDER BY total_wins DESC
LIMIT 1;

Explanation
------------
JOIN Matches : Connect Teams table with Matches table.
ON Teams.team_id = Matches.winner_team_id : Match team ids with winner team ids.
COUNT(Matches.match_id) : Count how many matches each team won.
GROUP BY Teams.team_name : Create groups for each team.
ORDER BY total_wins DESC : Sort by highest wins first.
LIMIT 1 : Show only the top team.

output
-------
+----------------+------------+
| team_name      | total_wins |
+----------------+------------+
| Mumbai Indians |          1 |
+----------------+------------+

4. Find the player who belongs to the team with the smallest team_id

query
------
SELECT player_name, role
FROM Players
WHERE team_id = (
    SELECT MIN(team_id)
    FROM Teams
);

Explanation
------------
SELECT player_name, role : Show the player name and role.
FROM Players : Take data from the Players table.
WHERE team_id = (...) : Select only players whose team_id matches the value returned by the subquery.

Subquery (Inner Query) :
SELECT MIN(team_id) : Find the smallest team_id.
FROM Teams : Use the Teams table.
MIN(team_id) returns 1.

Final Working :
The subquery returns 1.
The main query becomes:
SELECT player_name, role
FROM Players
WHERE team_id = 1;
This shows all players who belong to team_id 1, which is Mumbai Indians.

output
-------
+----------------+---------+
| player_name    | role    |
+----------------+---------+
| Rohit Sharma   | Batsman |
| Jasprit Bumrah | Bowler  |
+----------------+---------+
